    // counts[initiatorType][revState]. Reset after each report so counts are a
    // delta and repeated reports on a long-lived document cannot count the same
    // asset twice.
    var counts = {};
    var total = 0;
    // Not reset, so an asset is only ever counted once per document.
    var seen = {};
    var pending = null;

    function metaContent(name) {
      var tag = document.querySelector('meta[name="' + name + '"]');
      return tag ? tag.content : '';
    }

    function report() {
      pending = null;

      var apiUrl = metaContent('bb-api-canon-url');
      if (!apiUrl || !total) {
        return;
      }

      var counters = counts;
      counts = {};
      total = 0;

      var tags = [
        'source:bitbucket_core',
        'view_name:' + (metaContent('bb-view-name') || 'unknown')
      ];
      var payload = [];

      // One entry per initiator type and rev state, so both are dimensions that
      // can be grouped on rather than a separate metric name per combination.
      for (var type in counters) {
        if (!Object.prototype.hasOwnProperty.call(counters, type)) {
          continue;
        }
        var byRev = counters[type];
        for (var rev in byRev) {
          if (Object.prototype.hasOwnProperty.call(byRev, rev)) {
            payload.push({
              metric: 'increment',
              data: { 'bifrost.s3_asset': byRev[rev] },
              tags: tags.concat(['initiator_type:' + type, 'rev:' + rev])
            });
          }
        }
      }

      fetch(apiUrl + '/internal/metrics/statsd/', {
        method: 'POST',
        keepalive: true,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      })['catch'](function () {});
    }

    function record(entries) {
      for (var i = 0; i < entries.length; i++) {
        var entry = entries[i];
        for (var j = 0; j < origins.length; j++) {
          if (entry.name.indexOf(origins[j]) !== 0) {
            continue;
          }
          if (!seen[entry.name]) {
            seen[entry.name] = true;
            var type = entry.initiatorType || 'unknown';
            var rev = 'unknown';
            if (currentRev) {
              rev = entry.name.indexOf(currentRev) === -1 ? 'stale' : 'current';
            }
            var byRev = counts[type] || (counts[type] = {});
            byRev[rev] = (byRev[rev] || 0) + 1;
            total += 1;
            if (pending === null) {
              pending = window.setTimeout(report, REPORT_DELAY_MS);
            }
          }
          break;
        }
      }
    }

    try {
      // `buffered` replays the entries that loaded before this ran, so the
      // 250-entry resource timing buffer cannot drop them before we look.
      new window.PerformanceObserver(function (list) {
        record(list.getEntries());
      }).observe({ type: 'resource', buffered: true });
    } catch (e) {
      // A browser without buffered resource observation. The audit does not run.
    }
  })();
</script>
